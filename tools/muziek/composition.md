## 1. Algemene Track-informatie & Master-instellingen

* **Titel:** Mega-Volt Overdrive: Turbo Edition (Clean 16-Bit & Synthwave)
* **Genre:** High-Speed 16-Bit Arcade Synthwave / Clean Power-Rock (Sega Mega Drive / Genesis stijl)
* **Referenties:** *Thunder Force IV*, *Streets of Rage 2* (Yuzo Koshiro), *Sonic the Hedgehog*, *OutRun*

* **Maatsoort:** $\frac{4}{4}$ ($32\text{ maten} = 512\text{ zestiende-noot stappen}$)
* **Toonsoort:** E Mineur (E Aeolisch), met uitstapjes naar **E Frygisch** (`F5` voor eindbaas-spanning) en **E Harmonisch Mineur** (`B7` met `D#` in de climaxen)

## 6. Python-Script voor het 6-Sporen MIDI-Bestand (`mega_volt_overdrive_clean_156bpm.mid`)

Voer dit script uit met Python (`pip install midiutil`) om exact dezelfde $32$-maten multi-track MIDI te genereren als de ingebouwde export-knop in de Web Synthesizer:

~~~python
from midiutil import MIDIFile

def generate_turbo_clean_midi(filename="mega_volt_overdrive_clean_156bpm.mid"):
    # 6 Tracks: 0=FM Bass, 1=FM Lead, 2=Chiptune Arp, 3=Synth Brass, 4=Clean Power-Guitar, 5=Drums
    midi = MIDIFile(6)
    tempo = 156

    track_info = [
        ("YM2612 Aggressive FM Bass", 0, 38),
        ("Mega Drive Clean Saw Lead", 1, 81),
        ("SN76489 Turbo Pulse Arp",   2, 80),
        ("Nitro Synth-Brass Stabs",   3, 62),
        ("Clean 16-Bit Power-Guitar", 4, 27), # GM #27: Electric Guitar (clean)
        ("16-Bit Double-Kick Drums",  9, 0)
    ]

    for idx, (name, ch, prog) in enumerate(track_info):
        midi.addTrackName(idx, 0, name)
        midi.addTempo(idx, 0, tempo)
        if ch != 9:
            midi.addProgramChange(idx, ch, 0, prog)

    def add_step_ev(track_idx, ch, bar0, step16, pitches, dur_steps=1, vel=105):
        beat = bar0 * 4.0 + step16 * 0.25
        dur_beats = dur_steps * 0.25
        if not isinstance(pitches, list):
            pitches = [pitches]
        for p in pitches:
            midi.addNote(track_idx, ch, p, beat, dur_beats, vel)

    # Hulpfuncties voor begeleiding
    def add_gallop_bass(bar0, r_lo, r_hi, fifth, flat_six):
        pat = [r_lo, r_lo, r_hi, r_lo, fifth, r_lo, r_hi, r_lo,
               r_lo, r_lo, r_hi, r_lo, flat_six, fifth, r_lo, r_hi]
        for s, pitch in enumerate(pat):
            add_step_ev(0, 0, bar0, s, pitch, 1, 118 if s % 4 == 0 else 102)

    for b in range(32):
        # Drums basis (Kick=36, Snare=38, CHH=42, OHH=46)
        for s in range(16):
            add_step_ev(5, 9, b, s, 46 if s % 4 == 2 else 42, 1, 95 if s % 4 == 2 else 78)
            if s in (4, 12):
                add_step_ev(5, 9, b, s, 38, 2, 122)
            if s in (0, 2, 3, 6, 7, 8, 10, 11, 14, 15):
                add_step_ev(5, 9, b, s, 36, 1, 118)

    # Voorbeeld: Voeg de 16e-noot Twin-Turbo Solo (Maat 25-27) toe in tertsen!
    solo_runs = [
        (24, [76,74,72,71,72,74,76,74,72,71,69,71,72,71,69,67],
             [79,78,76,74,76,78,79,78,76,74,72,74,76,74,72,71]),
        (25, [71,72,74,76,74,72,74,76,78,76,74,72,74,76,78,79],
             [74,76,78,79,78,76,78,79,81,79,78,76,78,79,81,83]),
        (26, [78,76,74,76,74,72,71,72,71,69,67,69,71,72,74,76],
             [81,79,78,79,78,76,74,76,74,72,71,72,74,76,78,79])
    ]
    for bar0, gtr_notes, lead_notes in solo_runs:
        add_gallop_bass(bar0, 40, 52, 47, 48)
        for s in range(16):
            add_step_ev(4, 4, bar0, s, gtr_notes[s], 1, 118)
            add_step_ev(1, 1, bar0, s, lead_notes[s], 1, 114)

    with open(filename, "wb") as f:
        midi.writeFile(f)
    print(f"Gereed: {filename}")

if __name__ == "__main__":
    generate_turbo_clean_midi()
~~~
